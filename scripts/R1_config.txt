! =====================================================================
! R1 - Router Cisco IOL L3 - Asignacion 2 - AAA con RADIUS (NPS)
! Jose Gabriel Feliz Maria - 2025-0742
! Router entre las VLAN (router-on-a-stick por e0/0 hacia SW1):
!   VLAN 10 ADMINISTRACION 10.7.42.0/25   gw 10.7.42.1   (PC admin .20 por DHCP)
!   VLAN 20 SERVIDORES     10.7.42.128/29 gw 10.7.42.129 (Windows Server AD/NPS .130)
!   VLAN 99 GESTION        10.7.42.136/29 gw 10.7.42.137 (SW1 .138)
! =====================================================================

! ---------- PARTE 1: red basica ----------
enable
configure terminal
hostname R1
no ip domain-lookup
no service config
interface Ethernet0/0
 description TRUNK-SW1-e0/0
 no ip address
 no shutdown
interface Ethernet0/0.10
 description VLAN10-ADMINISTRACION
 encapsulation dot1Q 10
 ip address 10.7.42.1 255.255.255.128
interface Ethernet0/0.20
 description VLAN20-SERVIDORES
 encapsulation dot1Q 20
 ip address 10.7.42.129 255.255.255.248
interface Ethernet0/0.99
 description VLAN99-GESTION
 encapsulation dot1Q 99
 ip address 10.7.42.137 255.255.255.248
interface range Ethernet0/1 - 3
 description SIN-USO
 shutdown
exit
ip dhcp excluded-address 10.7.42.1 10.7.42.9
ip dhcp pool VLAN10-ADMINISTRACION
 network 10.7.42.0 255.255.255.128
 default-router 10.7.42.1
ip dhcp pool PC-ADMIN
 host 10.7.42.20 255.255.255.128
 client-identifier 0100.5056.c000.08
 default-router 10.7.42.1
exit
end
write memory

! ---------- PARTE 2: secretos (se escriben a mano, no se publican) ----------
! Usuario local de recuperacion: nombre poco obvio, clave fuerte, privilegio 15.
! Solo sirve por consola y auxiliar, y solo si el NPS no responde.
configure terminal
security passwords min-length 10
! Primero tipo 5 y luego scrypt (tipo 9), que la reemplaza; si el IOS no acepta scrypt, queda la tipo 5.
enable secret <CLAVE_ENABLE>
enable algorithm-type scrypt secret <CLAVE_ENABLE>
username <USUARIO_RECUPERACION> privilege 15 secret <CLAVE_RECUPERACION>
username <USUARIO_RECUPERACION> privilege 15 algorithm-type scrypt secret <CLAVE_RECUPERACION>
end

! ---------- PARTE 3: AAA, RADIUS, SSH, niveles de privilegio y registros ----------
configure terminal
service password-encryption
service timestamps log datetime msec localtime show-timezone
service timestamps debug datetime msec localtime show-timezone
clock timezone AST -4 0
ntp server 10.7.42.130
banner motd #R1 - Acceso restringido a personal autorizado. AAA con RADIUS - 2025-0742#
ip domain-name jose.gabriel
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
ip ssh logging events
access-list 10 remark Solo la VLAN 10 (administracion) abre SSH
access-list 10 permit 10.7.42.0 0.0.0.127
! --- RADIUS como cliente del NPS ---
aaa new-model
radius server NPS
 address ipv4 10.7.42.130 auth-port 1812 acct-port 1813
 timeout 5
 retransmit 2
exit
aaa group server radius GRUPO-NPS
 server name NPS
 ip radius source-interface Ethernet0/0.20
exit
! --- Autenticacion: SSH solo por RADIUS; consola y auxiliar RADIUS y, si no responde, el usuario local ---
aaa authentication login default group GRUPO-NPS
aaa authentication login VTY-RADIUS group GRUPO-NPS
aaa authentication login CON-AUX group GRUPO-NPS local
aaa authentication enable default enable
! --- Autorizacion: el NPS entrega el nivel (shell:priv-lvl) ---
aaa authorization console
aaa authorization exec default group GRUPO-NPS
aaa authorization exec VTY-RADIUS group GRUPO-NPS
aaa authorization exec CON-AUX group GRUPO-NPS local
! --- Registro de sesiones en el NPS ---
aaa accounting exec default start-stop group GRUPO-NPS
! --- Que puede hacer cada nivel ---
privilege exec level 0 show privilege
privilege exec level 5 show ip interface brief
privilege exec level 5 show ip route
privilege exec level 5 show interfaces
privilege exec level 5 show logging
privilege exec level 5 show users
privilege exec level 5 ping
privilege exec level 5 traceroute
privilege exec level 10 configure terminal
privilege configure level 10 interface
privilege interface level 10 description
privilege interface level 10 shutdown
privilege exec level 10 show running-config
! IOS sube la palabra 'show' al nivel de cada 'show ...' asignado (aqui quedaba en 10);
! se devuelve 'show' a nivel 0 al final para que los niveles 0 y 5 puedan usar sus show.
privilege exec level 0 show
privilege exec level 0 show privilege
! --- Registros de autenticacion, autorizacion y cambios ---
login block-for 60 attempts 5 within 60
login on-success log
login on-failure log
logging buffered 64000 debugging
archive
 log config
  logging enable
  notify syslog contenttype plaintext
  hidekeys
exit
exit
! --- Lineas ---
line con 0
 login authentication CON-AUX
 authorization exec CON-AUX
 exec-timeout 10 0
 logging synchronous
line aux 0
 login authentication CON-AUX
 authorization exec CON-AUX
 exec-timeout 10 0
line vty 0 4
 login authentication VTY-RADIUS
 authorization exec VTY-RADIUS
 access-class 10 in
 transport input ssh
 exec-timeout 10 0
end
write memory

! ---------- PARTE 4: secreto RADIUS (a mano, no se publica) ----------
configure terminal
radius server NPS
 key <SECRETO_RADIUS>
end
write memory
